## Usage

```tsx
import {
  Timeline,
  TimelineItem,
  TimelineBullet,
  TimelineConnector,
  TimelineContent
} from "@/components/marval-ui/timeline";
```

```tsx
<Timeline>
  <TimelineItem>
    <TimelineBullet />
    <TimelineConnector />
    <TimelineContent />
  </TimelineItem>
</Timeline>
```
