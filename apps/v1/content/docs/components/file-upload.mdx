```tsx
"use client"

import { CloudUploadIcon } from "lucide-react"
import { useState } from "react"
import { toast } from "sonner"

import { useComponentPreviewDemoContext } from "@/components/shared/component-preview"
import { Button } from "@/registry/button"
import {
	FileUpload,
	FileUploadDropzone,
	FileUploadItem,
	FileUploadList,
	FileUploadTrigger
} from "@/registry/file-upload"

export function FileUploadDemo() {
	const [files, setFiles] = useState<File[]>([])

	const handleRejectFile = (file, error) => {
		if (error) {
			const reason = error.reason
			if (reason === "MAX_SIZE_REACHED") {
				toast.error(`${error.message}`, {
					duration: 1000 * 10
				})
			}
		}
	}

	return (
		<div className="mt-20 w-full max-w-md">
			<FileUpload
				value={files}
				onValueChange={setFiles}
				onRejectFile={handleRejectFile}
			>
				<FileUploadDropzone className="flex-row">
					<CloudUploadIcon className="size-4" />
					Drag and drop or
					<FileUploadTrigger asChild>
						<Button variant="link" size="sm" className="p-0">
							choose files
						</Button>
					</FileUploadTrigger>
					to upload
				</FileUploadDropzone>

				<FileUploadList>
					{files.map((file) => (
						<FileUploadItem key={file.name} />
					))}
				</FileUploadList>
			</FileUpload>
		</div>
	)
}
```

## Usage

```tsx
import {
	FileUpload,
	FileUploadDropzone,
	FileUploadTrigger,
	FileUploadList,
	FileUploadItem,
	FileUploadItemPreview,
	FileUploadItemMetadata,
	FileUploadItemName,
	FileUploadItemSize,
	FileUploadItemStatus,
	FileUploadItemError,
	FileUploadItemProgressWithLabel,
	FileUploadItemProgress,
	FileUploadItemProgressLabel,
	FileUploadItemDelete,
	FileUploadItemRetry,
	FileUploadItemMedia,
} from "@/components/marval-ui/file-upload"
```

```tsx
return (
	<FileUpload>
		<FileUploadDropzone>
			<FileUploadMedia />
		</FileUploadDropzone>
		<FileUploadTrigger />
		<FileUploadList>
			<FileUploadItem>
				<FileUploadItemPreview />
				<FileUploadItemMetadata />
				<FileUploadItemProgress />
				<FileUploadItemDelete />
			</FileUploadItem>
		</FileUploadList>
	</FileUpload>
)
```
